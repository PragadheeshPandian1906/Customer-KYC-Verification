$bytes = New-Object byte[] 32
$rng = [System.Security.Cryptography.RandomNumberGenerator]::Create()
try {
    $rng.GetBytes($bytes)
    [Convert]::ToBase64String($bytes) |
        Set-Clipboard
     Write-Host "Generated a 32-byte key and copied it to the clipboard."
}
finally {
    $rng.Dispose()
}
